α task

As an optional task create a new violin plot that includes the following:

  • Plots the metrics:
    • “diversity_inverse_simpson” renaming it as “invsimpson”
    • “evenness_pielou” renaming it as “pielou”
    • “diversity_fisher” renaming it as “fisher”
    • Ensure they are in the order “pielou”, “invsimpson”, then “fisher”
  • The x-axis is separated by Site (UD, MD, LD) rather than media.
    • Ensure the order is UD, MD, then LD
  • Points are coloured by Media (ENV, CVP, KBC, and TSA)
    • Ensure the order is ENV, CVP, KBC, then TSA
  • Include Wilcoxon paired stats comparing the Sites
Note

Going to earlier code, renaming, adding new options, and rerunning the entire notebook may be useful.

NoteData frame with metrics of choice

Subset long alpha diversity table created earlier in this section to contain metrics of choice.

#To compare means we need to create a list of comparisons
#Create character vector of unique metadata values (media in this case)
uniq_media_values_chr_vec <- basic_alpha_metrics_tbl |>
    #Extract media as a vector
    dplyr::pull(media) |>
    #Convert to character
    as.character() |>
    #Unique values
    unique()
#View vector
uniq_media_values_chr_vec
NoteCreate metadata combination list for plot stats 
#To compare means we need to create a list of comparisons
#Create list of site 2-way comparisons
my_comparisons <- subset_alpha_metrics_tbl |>
    #Extract site column as a vector
    dplyr::pull(site) |>
    #Convert to character
    as.character() |>
    #Unique values
    unique() |>
    #Pairwise combinations of sites
    combn(m =2, simplify=FALSE)
#Check contents and structure
my_comparisons
str(my_comparisons)
NoteReorder factors and comparisons 
#Reorder sites, media and comparisons
#Set order of sites
subset_alpha_metrics_tbl$site <- factor(subset_alpha_metrics_tbl$site,
                                           #set order of levels
                                           levels = c("UD", "MD", "LD"))
#Set order of media
subset_alpha_metrics_tbl$media <- factor(subset_alpha_metrics_tbl$media,
                                            #set order of levels
                                            levels = c("ENV","CVP","KBC","TSA"))
#Order comparisons
my_ordered_comparisons <- my_comparisons[c(3,1,2)]
my_ordered_comparisons
NotePlot with stats 
#Produce ggplot object of violin plot
alpha_violinplot <- ggplot(subset_alpha_metrics_tbl, aes(x = site, y = value)) +
                            ggplot2::geom_violin() +
                            ggforce::geom_sina(alpha=0.5, aes(colour=media)) +
                            ggplot2::labs(colour = "Media", x = "Site", y = "Value") +
                            ggplot2::facet_wrap(~metric, scales = "free") +
                            #Add comparisons
                            ggpubr::stat_compare_means(comparisons = my_ordered_comparisons)
#Save ggplot2 object with ggsave
ggsave(filename = "./Alpha_diversity_rarefy_iters_site_violinplot_pairwise_wilcox.png", plot = alpha_violinplot,
      device = "png", dpi = 300, units = "mm", height = 200, width = 250)
#Display plot
IRdisplay::display_png(file = "./Alpha_diversity_rarefy_iters_site_violinplot_pairwise_wilcox.png")
